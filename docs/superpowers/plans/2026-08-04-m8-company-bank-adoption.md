from pathlib import Path

from src.company_bank.model import FactKind, SourceKind
from src.company_bank.serde import load_seed_companies
from src.company_bank.store import load_company_bank

SEEDS = Path("config/company_bank/seed_companies.yaml")
ROOT = Path("config/company_bank")


def test_seed_corpus_is_exactly_complete():
    seeds = load_seed_companies(SEEDS)
    bank = load_company_bank(ROOT)
    assert len(seeds) == 30
    assert set(bank.dossiers) == set(seeds)
    assert {
        company_id: dossier.display_name
        for company_id, dossier in bank.dossiers.items()

def test_every_company_has_official_product_domain_or_engineering_evidence():
    bank = load_company_bank(ROOT)
    qualifying = {FactKind.PRODUCT, FactKind.DOMAIN, FactKind.ENGINEERING_THEME}
    for company_id, dossier in bank.dossiers.items():
        assert any(source.source_kind is not SourceKind.VERIFIED_DATASET for source in dossier.sources), company_id
        assert any(fact.kind in qualifying for fact in dossier.facts), company_id


def test_hiring_guidance_uses_only_official_hiring_sources():
    bank = load_company_bank(ROOT)
    allowed = {SourceKind.OFFICIAL_CAREERS, SourceKind.OFFICIAL_HIRING_GUIDE}
    for company_id, dossier in bank.dossiers.items():
        sources = {source.id: source for source in dossier.sources}
        for fact in dossier.facts:
            if fact.kind is FactKind.HIRING_GUIDANCE:
                assert sources[fact.source_id].source_kind in allowed, (company_id, fact.id)


def test_every_signal_has_a_resolved_basis_fact():
    bank = load_company_bank(ROOT)
    for company_id, dossier in bank.dossiers.items():
        fact_ids = {fact.id for fact in dossier.facts}
        for signal in dossier.signals:
            assert signal.basis_fact_ids, (company_id, signal.id)
            assert set(signal.basis_fact_ids) <= fact_ids, (company_id, signal.id)


def test_canonical_yaml_contains_no_prohibited_field_names():
    prohibited = {
        "candidate_name", "email", "phone", "resume", "master_profile",
        "sponsorship", "citizenship", "visa", "work_authorization",
        "location_eligibility", "start_window",
    }
    for path in sorted((ROOT / "companies").glob("*.yaml")):
        text = path.read_text(encoding="utf-8").casefold()
        for field in prohibited:
            assert f"{field}:" not in text, (path.name, field)

- [ ] **Step 3: Implement pure aggregation and rendering**

Define this frozen dataclass in `src/company_bank/report.py`:

```python
@dataclass(frozen=True)
class CompanyCoverage:
    company_id: str
    display_name: str
    source_count: int
    fact_counts: tuple[tuple[str, int], ...]
    signal_count: int
    scopes: tuple[str, ...]
    researched_at: str
    expires_at: str


def build_coverage(bank: CompanyBank) -> tuple[CompanyCoverage, ...]:
    rows: list[CompanyCoverage] = []
    for company_id in sorted(bank.dossiers):
        dossier = bank.dossiers[company_id]
        counts = Counter(fact.kind.value for fact in dossier.facts)
        scopes = {
            f"{scope.kind.value}:{scope.name}"
            for scope in (
                *(source.scope for source in dossier.sources),
                *(fact.scope for fact in dossier.facts),
            )
        }
        rows.append(CompanyCoverage(
            company_id=company_id,
            display_name=dossier.display_name,
            source_count=len(dossier.sources),
            fact_counts=tuple(sorted(counts.items())),
            signal_count=len(dossier.signals),
            scopes=tuple(sorted(scopes)),
            researched_at=format_utc_timestamp(dossier.researched_at),
            expires_at=format_utc_timestamp(dossier.expires_at),
        ))
    return tuple(rows)


def render_coverage_markdown(rows: tuple[CompanyCoverage, ...]) -> str:
    total_sources = sum(row.source_count for row in rows)
    total_facts = sum(sum(count for _, count in row.fact_counts) for row in rows)
    total_signals = sum(row.signal_count for row in rows)
    lines = [
        "# Company Knowledge Bank v0.1.0 Coverage",
        "",
        f"Companies: {len(rows)}; sources: {total_sources}; facts: {total_facts}; signals: {total_signals}.",
        "",
        "| ID | Company | Sources | Facts by kind | Signals | Scopes | Researched | Expires |",
        "|---|---|---:|---|---:|---|---|---|",
    ]
    for row in rows:
        facts = ", ".join(f"{kind}={count}" for kind, count in row.fact_counts)
        scopes = ", ".join(row.scopes)
        lines.append(
            f"| {row.company_id} | {row.display_name} | {row.source_count} | "
            f"{facts} | {row.signal_count} | {scopes} | "
            f"{row.researched_at} | {row.expires_at} |"
        )
    return "\n".join(lines) + "\n"
```
