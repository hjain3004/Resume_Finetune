\documentclass[letterpaper,11pt]{article}

\usepackage{latexsym}
% ATS text extraction depends on all three of these together:
%   lmodern  - Type1 fonts with real glyph names, so the fi/ff/fl ligatures
%              extract as letters instead of (cid:28)/(cid:27)/(cid:29).
%   T1       - gives \textasciitilde an ASCII "~" (OT1 yields U+02DC SMALL
%              TILDE), which is what keeps the "~40%" metric hedging readable.
%   textcomp - provides \textquotesingle, extracting as ASCII "'" rather than
%              the U+2019 curly apostrophe that ats.forbidden_chars rejects.
\usepackage{lmodern}
\usepackage[T1]{fontenc}
\usepackage{textcomp}

  left=0.20in, right=0.20in,
